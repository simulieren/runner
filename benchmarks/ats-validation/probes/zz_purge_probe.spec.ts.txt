// Temporary lab probe; removed before full-suite comparisons.
import db from '@adonisjs/lucid/services/db'
import { test } from '@japa/runner'
import { DateTime } from 'luxon'

import PurgeSoftDeletedJob from '#jobs/purge_soft_deleted_job'

import crypto from 'node:crypto'

test('probe: one failing user does not block the rest of its chunk', async ({ assert }) => {
  await db.from('users').whereNotNull('deleted_at').delete()
  const cutoff = DateTime.now().minus({ days: 31 }).toSQL()!
  const ids = Array.from({ length: 5 }, () => crypto.randomUUID())
  await db
    .table('users')
    .multiInsert(
      ids.map((id, i) => ({
        id,
        full_name: `Probe ${i}`,
        email: `probe-${id}@example.com`,
        password: 'x',
        role: 'user',
        deleted_at: cutoff,
        created_at: cutoff,
        updated_at: cutoff,
      }))
    )
  const logId = crypto.randomUUID()
  await db
    .table('admin_action_logs')
    .insert({
      id: logId,
      admin_user_id: ids[2],
      action: 'probe',
      target_type: 'probe',
      target_id: logId,
      created_at: cutoff,
    })
  await db.rawQuery(`CREATE FUNCTION zz_probe_block() RETURNS trigger LANGUAGE plpgsql AS $$
    BEGIN IF OLD.id = '${ids[2]}' THEN RAISE EXCEPTION 'probe block'; END IF; RETURN OLD; END $$`)
  await db.rawQuery(
    'CREATE TRIGGER zz_probe_block BEFORE DELETE ON users FOR EACH ROW EXECUTE FUNCTION zz_probe_block()'
  )
  try {
    const job = new PurgeSoftDeletedJob()
    const summary = await (
      job as unknown as { runPurge(cutoff: Date): Promise<{ usersPurged: number; errors: number }> }
    ).runPurge(DateTime.now().minus({ days: 30 }).toJSDate())
    const left = await db.from('users').whereIn('id', ids).select('id')
    const log = await db.from('admin_action_logs').where('id', logId).first()
    console.log(
      `[purge-fallback] ${JSON.stringify({ summary, left: left.map((r) => ids.indexOf(r.id)), logUser: log.admin_user_id === ids[2], label: log.deleted_user_label })}`
    )
    assert.deepEqual(
      left.map((r) => r.id),
      [ids[2]]
    )
    assert.equal(summary.errors, 1)
    assert.equal(log.admin_user_id, ids[2])
  } finally {
    await db.rawQuery('DROP TRIGGER zz_probe_block ON users')
    await db.rawQuery('DROP FUNCTION zz_probe_block()')
    await db.from('admin_action_logs').where('id', logId).delete()
    await db
      .from('users')
      .whereIn('id', ids)
      .delete()
      .catch(() => {})
  }
})
